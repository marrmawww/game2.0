cmake_minimum_required(VERSION 3.10)
project(Lab2 CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
include_directories(${CMAKE_CURRENT_SOURCE_DIR}/app ${CMAKE_CURRENT_SOURCE_DIR}/core ${CMAKE_CURRENT_SOURCE_DIR}/units ${CMAKE_CURRENT_SOURCE_DIR}/factories ${CMAKE_CURRENT_SOURCE_DIR}/terrain ${CMAKE_CURRENT_SOURCE_DIR}/neutral ${CMAKE_CURRENT_SOURCE_DIR}/patterns ${CMAKE_CURRENT_SOURCE_DIR}/logging)
add_library(game logging/Logger.cpp units/Unit.cpp units/UnitKind.cpp factories/UnitFactory.cpp terrain/TerrainProxy.cpp terrain/TerrainMap.cpp neutral/Strategies.cpp core/Base.cpp core/GameController.cpp)
target_link_libraries(game Threads::Threads)
add_executable(lab2 app/main.cpp)
target_link_libraries(lab2 game)
add_executable(lab2_tests tests/tests.cpp)
target_link_libraries(lab2_tests game)
